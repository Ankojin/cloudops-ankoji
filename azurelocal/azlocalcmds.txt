  $vms = @()

#shows both control-plane and node-pool VMs across the entire Azure Stack HCI cluster

Get-ClusterNode | ForEach-Object {
    $node = $_.Name
    $vms += Invoke-Command -ComputerName $node -ScriptBlock {
        Get-VM | Where-Object { 
            $_.Name -like "*control-plane*" -or 
            $_.Name -like "*nodepool*" -or 
            $_.Name -like "*aks*" 
        } | Select-Object Name, State, 
            @{N='MemoryGB';E={[math]::Round($_.MemoryAssigned/1GB,1)}},
            Uptime,
            @{N='Host';E={$env:COMPUTERNAME}}
    }
}

$vms | Sort-Object Name | Format-Table -AutoSize
